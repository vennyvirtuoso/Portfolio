kernels/soft_dice_loss.py 2D Block Grid + Lock-Free tl.atomic_add 
import torch
import triton
import triton.language as tl

# Fused Soft-Dice Loss Kernel for 3D/2D Tensors
# Solves standard Triton tutorial block SRAM limitations by partitioning 
# spatial elements into chunks (Axis 0) and accumulating across slices (Axis 1).
@triton.jit
def _soft_dice_kernel(
    pred_ptr, target_ptr,
    inter_ptr, pred_sum_ptr, target_sum_ptr,  # Output reduction buffers
    n_elements,                              # Elements per (B, C) slice (~260K+ elements)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)      # Axis 0: Spatial chunk index
    row_id = tl.program_id(1)   # Axis 1: Batch * Channel slice index

    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    base = row_id * n_elements
    pred = tl.load(pred_ptr + base + offsets, mask=mask, other=0.0)
    target = tl.load(target_ptr + base + offsets, mask=mask, other=0.0)

    # Compute partial sums in SRAM registers without global DRAM roundtrips
    inter = tl.sum(pred * target, axis=0)
    p_sum = tl.sum(pred, axis=0)
    t_sum = tl.sum(target, axis=0)

    # Lock-free atomic accumulation across cooperating spatial blocks
    tl.atomic_add(inter_ptr + row_id, inter)
    tl.atomic_add(pred_sum_ptr + row_id, p_sum)
    tl.atomic_add(target_sum_ptr + row_id, t_sum)
kernels/fused_softmax.py Single-Pass Row Softmax in Registers 
@triton.jit
def _softmax_kernel(
    out_ptr, in_ptr,
    in_row_stride, out_row_stride,
    n_cols,
    BLOCK_SIZE: tl.constexpr,  # Pow2 padded block size
):
    row_idx = tl.program_id(0)
    row_start_ptr = in_ptr + row_idx * in_row_stride
    col_offsets = tl.arange(0, BLOCK_SIZE)
    input_ptrs = row_start_ptr + col_offsets
    mask = col_offsets < n_cols

    # Load entire row into SRAM registers
    row = tl.load(input_ptrs, mask=mask, other=-float("inf"))

    # Perform max, sub, exp, sum, div entirely in registers
    row_minus_max = row - tl.max(row, axis=0)
    numerator = tl.exp(row_minus_max)
    denominator = tl.sum(numerator, axis=0)
    softmax_out = numerator / denominator

    # Single writeback to global memory
    out_row_start_ptr = out_ptr + row_idx * out_row_stride
    tl.store(out_row_start_ptr + col_offsets, softmax_out, mask=mask)
kernels/vector_add.py Triton Grid, Program IDs & Masking Baseline 
@triton.jit
def _vector_add_kernel(
    x_ptr, y_ptr, z_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    x = tl.load(x_ptr + offsets, mask=mask)
    y = tl.load(y_ptr + offsets, mask=mask)
    output = x + y
    tl.store(z_ptr + offsets, output, mask=mask)
Architecture & Hardware

GPU Memory Hierarchy & Optimization Strategy

Why custom Triton kernels outpace eager framework execution: eliminating bandwidth bottlenecks and optimizing for modern GPU memory architecture.

⚡

Kernel Fusion vs Eager Overhead

Standard PyTorch eager execution launches 4 sequential ATen kernels for Soft-Dice Loss: elementwise multiplication and 3 separate .sum() reduction passes. This forces 4 complete HBM global memory roundtrips. By fusing elementwise multiplication and partial reductions into a single Triton kernel launch, DRAM reads/writes are cut by ~75%.

HBM Bandwidth Reduction
🔲

2D Grid & Atomic Accumulation

Medical segmentation tensors (e.g. 512 × 512 × 128) contain ~262K+ spatial elements per slice, exceeding standard SRAM block registers. We architect a 2D block grid: Axis 0 tiles spatial chunks across blocks, while Axis 1 indexes slice channels. Cooperating blocks atomically reduce partial sums into final output scalars via lock-free tl.atomic_add.

2D Grid Partitioning
🧠

Coalesced Vectorized Memory Access

Utilizing vectorized loads (tl.load) aligned to power-of-two block sizes guarantees memory transactions saturate GPU warp memory coalescing (128-byte transactions). Intermediate reduction values remain pinned inside fast L1 cache and register files before writing to global memory.
